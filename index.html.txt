<script>
  (function() {
    // ----- state -----
    const MAX_ADS = 20;
    const TOTAL_SPINS = 20;

    let watchedAds = 0;
    let spinsUsed = 0;
    let spinsUnlocked = 0;

    // DOM elements
    const adProgressFill = document.getElementById('adProgressFill');
    const adCounter = document.getElementById('adCounter');
    const watchBtn = document.getElementById('watchAdBtn');
    const spinBtn = document.getElementById('spinBtn');
    const resultDisplay = document.getElementById('resultDisplay');
    const wheel = document.getElementById('wheel');
    const wheelCenter = document.getElementById('wheelCenter');
    const spinAttemptsSpan = document.getElementById('spinAttempts');
    const resetTimerSpan = document.getElementById('resetTimer');
    const unlockStatus = document.getElementById('unlockStatus');
    const spinCounter = document.getElementById('spinCounter');

    // ----- helper: update UI -----
    function refreshUI() {
      const percent = (watchedAds / MAX_ADS) * 100;
      adProgressFill.style.width = Math.min(percent, 100) + '%';
      adCounter.textContent = `${watchedAds} / ${MAX_ADS}`;

      if (watchedAds >= MAX_ADS) {
        watchBtn.disabled = true;
        watchBtn.innerHTML = '✅ UNLOCKED';
      } else {
        watchBtn.disabled = false;
        watchBtn.innerHTML = '▶ WATCH AD';
      }

      if (watchedAds >= MAX_ADS) {
        spinsUnlocked = TOTAL_SPINS;
        unlockStatus.textContent = '🔓 20/20';
        unlockStatus.className = 'unlock-badge unlocked';
      } else {
        spinsUnlocked = 0;
        unlockStatus.textContent = `🔒 ${watchedAds}/20`;
        unlockStatus.className = 'unlock-badge';
      }

      let remainingSpins = 0;
      if (spinsUnlocked === TOTAL_SPINS) {
        remainingSpins = TOTAL_SPINS - spinsUsed;
        if (remainingSpins < 0) remainingSpins = 0;
      }

      spinCounter.textContent = remainingSpins;
      spinAttemptsSpan.textContent = `${remainingSpins} spin${remainingSpins !== 1 ? 's' : ''} left`;

      if (spinsUnlocked === TOTAL_SPINS && remainingSpins > 0) {
        spinBtn.disabled = false;
      } else {
        spinBtn.disabled = true;
      }
    }

    // ----- watch ad (triggers SDK if available) -----
    function watchAd() {
      if (watchedAds >= MAX_ADS) return;

      // Try to show ad via libtl SDK
      if (typeof show_11485825 !== 'undefined') {
        try {
          show_11485825();
        } catch (e) {
          console.warn('libtl ad error:', e);
        }
      } else {
        console.warn('libtl SDK not loaded yet');
      }

      // Increment after ad (in real scenario, this would be called after ad completion)
      // For demo, we increment immediately. In production, you'd use a callback.
      watchedAds++;
      refreshUI();
    }

    // ----- spin logic -----
    function performSpin() {
      if (spinsUnlocked !== TOTAL_SPINS) return;
      const remaining = TOTAL_SPINS - spinsUsed;
      if (remaining <= 0) return;

      spinsUsed++;
      refreshUI();

      const result = Math.floor(Math.random() * 10) + 1;
      resultDisplay.textContent = result;
      wheelCenter.textContent = result;

      const sliceIndex = result - 1;
      const targetAngle = sliceIndex * 36 + 18;
      const extraTurns = 5 + Math.floor(Math.random() * 4);
      const rotation = extraTurns * 360 + targetAngle;
      wheel.style.transition = 'transform 1.2s cubic-bezier(0.34, 1.56, 0.64, 1)';
      wheel.style.transform = `rotate(${rotation}deg)`;
    }

    // ----- reset timer -----
    function updateResetTimer() {
      const now = new Date();
      const resetTime = new Date(now);
      resetTime.setHours(2, 0, 0, 0);
      if (now > resetTime) {
        resetTime.setDate(resetTime.getDate() + 1);
      }
      const diffMs = resetTime - now;
      if (diffMs <= 0) {
        resetTimerSpan.textContent = '00:00:00';
        return;
      }
      const totalSec = Math.floor(diffMs / 1000);
      const hours = String(Math.floor(totalSec / 3600)).padStart(2, '0');
      const minutes = String(Math.floor((totalSec % 3600) / 60)).padStart(2, '0');
      const seconds = String(totalSec % 60).padStart(2, '0');
      resetTimerSpan.textContent = `${hours}:${minutes}:${seconds}`;
    }

    // ----- reset daily -----
    function resetDaily() {
      watchedAds = 0;
      spinsUsed = 0;
      spinsUnlocked = 0;
      wheel.style.transition = 'none';
      wheel.style.transform = 'rotate(0deg)';
      wheelCenter.textContent = '?';
      resultDisplay.textContent = '--';
      refreshUI();
    }

    // ----- events -----
    watchBtn.addEventListener('click', function(e) {
      e.preventDefault();
      watchAd();
    });

    spinBtn.addEventListener('click', function(e) {
      e.preventDefault();
      performSpin();
    });

    // ----- init -----
    function init() {
      watchedAds = 0;
      spinsUsed = 0;
      spinsUnlocked = 0;
      refreshUI();
      resultDisplay.textContent = '--';
      wheelCenter.textContent = '?';
      wheel.style.transition = 'none';
      wheel.style.transform = 'rotate(0deg)';
      void wheel.offsetHeight;
      wheel.style.transition = 'transform 0.5s cubic-bezier(0.34, 1.56, 0.64, 1)';

      updateResetTimer();
      setInterval(updateResetTimer, 1000);

      setInterval(() => {
        if (resetTimerSpan.textContent === '00:00:00') {
          resetDaily();
        }
      }, 5000);

      // Move the SDK ad into our container
      const sdkContainer = document.getElementById('adSdkContainer');
      if (sdkContainer) {
        // The SDK might inject into body, we try to relocate it.
        setTimeout(() => {
          const sdkElements = document.querySelectorAll('[data-zone="11485825"]');
          sdkElements.forEach(el => {
            if (el.parentNode !== sdkContainer) {
              sdkContainer.appendChild(el);
            }
          });
          // Also look for any iframe or div that libtl might have created
          const adDivs = document.querySelectorAll('div[style*="z-index"]');
          adDivs.forEach(el => {
            if (el.innerHTML.includes('libtl') || el.id.includes('libtl')) {
              if (el.parentNode !== sdkContainer) {
                sdkContainer.appendChild(el);
              }
            }
          });
        }, 500);
      }
    }

    init();

    // double-click on bot name to manually reset (for testing)
    document.querySelector('.bot-name').addEventListener('dblclick', function() {
      resetDaily();
    });

  })();
</script>
